## インターフェース

```python
# src/so101_vlm/eval/geometry.py (新規)
@dataclass(frozen=True)
class SceneGeometry:
    block_names: tuple[str, ...]; block_half_size: float; block_rest_z: float        # 🔵
    box_inner_half_size: float; box_wall_thickness: float                            # 🔵 XML から
    box_floor_top_z: float; box_rim_z: float                                         # 🔵 XML から
    box_default_xy: tuple[float, float]                                              # 🔵 <body name="box" pos>
    camera_name: str; image_width: int; image_height: int                            # 🔵 XML 外の Python 定数
    @classmethod
    def from_xml(cls, path: Path = PICKPLACE_SCENE_PATH) -> "SceneGeometry": ...     # 🟡 ElementTree。メッシュを読まない
    @classmethod
    def from_model(cls, model: mujoco.MjModel) -> "SceneGeometry": ...               # 🟡 test_scene の照合用
def get_geometry() -> SceneGeometry: ...                                             # 🟡 lru_cache

@dataclass(frozen=True)
class PlaceRegion:                                                                   # 🔵 目標と判定で共有
    center_xy: tuple[float, float]; inner_half_size: float; rim_z: float; floor_top_z: float
    @property
    def goal_xyz(self) -> tuple[float, float, float]: ...                            # (cx, cy, floor_top_z)
```
